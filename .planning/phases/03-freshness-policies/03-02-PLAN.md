"""MetadataDBHook — Airflow metadata DB access for the aircontroler_model_runs state table.

FRESH-04: runtime reads and writes use `create_session` from `airflow.utils.session`
(same import path on Airflow 2.7+ and 3.x per RESEARCH Pitfall 1). This satisfies
the CLAUDE.md rule that workers must not open raw SQLAlchemy sessions from execute().

Security (T-3-01): all queries use SQLAlchemy ORM Core expressions against the
`model_runs_table` imported from aircontroler.db_manager — unique_id is bound as a
parameter, never formatted into a SQL string.
"""
from __future__ import annotations

from datetime import datetime, timezone

import sqlalchemy as sa
from airflow.hooks.base import BaseHook
from airflow.utils.session import create_session


class MetadataDBHook(BaseHook):
    """Hook for aircontroler state table access via Airflow's metadata DB session."""

    conn_name_attr = "aircontroler_conn_id"
    default_conn_name = "aircontroler_default"
    conn_type = "aircontroler"
    hook_name = "Aircontroler"

    def get_conn(self):
        """Not supported — MetadataDBHook uses create_session for all DB access."""
        raise NotImplementedError(
            "MetadataDBHook uses create_session (airflow.utils.session), not get_conn()."
        )

    def get_last_run(self, unique_id: str) -> datetime | None:
        """Return MAX(completed_at) for the given unique_id, or None if never recorded.

        FRESH-04 / T-3-01: uses parameterized ORM `sa.select(...).where(col == value)`.
        The unique_id is bound as a parameter; no string formatting is used.
        """
        from aircontroler.db_manager import model_runs_table
        with create_session() as session:
            result = session.execute(
                sa.select(sa.func.max(model_runs_table.c.completed_at))
                .where(model_runs_table.c.unique_id == unique_id)
            ).scalar()
        return result

    def record_run(self, unique_id: str, dag_id: str, run_id: str) -> None:
        """Insert a success row with completed_at=now(UTC) for this model run.

        D-04: append-only — never updates in place. D-05: writes all four non-id columns.
        """
        from aircontroler.db_manager import model_runs_table
        with create_session() as session:
            session.execute(
                model_runs_table.insert().values(
                    unique_id=unique_id,
                    dag_id=dag_id,
                    run_id=run_id,
                    completed_at=datetime.now(tz=timezone.utc),
                )
            )
```

Critical notes:
- Import `create_session` directly from `airflow.utils.session` — NOT from any common-compat shim (compat shim does not expose it; live-env verified).
- `model_runs_table` is imported lazily inside methods so the hook module can be imported before db_manager is reached (avoids any circular import from `__init__.py` additions in Plan 03).
- `session.execute(...)` for INSERT — `create_session` commits on `__exit__` when no exception is raised; no explicit `session.commit()` needed.

            if last_run is not None:
                # Pitfall 5: normalize to UTC. SQLite and some drivers strip tzinfo;
                # if naive, treat as UTC (matches the hook's write-side UTC stamp).
                if last_run.tzinfo is None:
                    last_run = last_run.replace(tzinfo=timezone.utc)
                now = datetime.now(tz=timezone.utc)
                elapsed_hours = (now - last_run).total_seconds() / 3600
                if elapsed_hours < self.freshness_hours:
                    self.log.info(
                        "Skipping %s: last run %.2f hours ago (TTL: %.1f hours)",
                        self.unique_id, elapsed_hours, self.freshness_hours,
                    )
                    raise AirflowSkipException(
                        f"{self.unique_id} is fresh "
                        f"(last ran {elapsed_hours:.2f}h ago, TTL {self.freshness_hours}h)"
                    )

        # Existing Phase 1 / 2 subprocess logic — unchanged structure
        executable = self.dbt_executable_path or "dbt"
        cmd = [
            executable,
            "build",
            "--select", self.unique_id,
            "--project-dir", self.project_dir,
            "--profiles-dir", self.profiles_path,
        ]
        self.log.info("Running dbt: %s", " ".join(cmd))
        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            check=False,
        )
        if result.returncode != 0:
            if result.stdout:
                self.log.error("dbt stdout:\n%s", result.stdout)
            self.log.error("dbt stderr:\n%s", result.stderr)
            raise AirflowException(
                f"dbt build failed for {self.unique_id!r} "
                f"(exit {result.returncode})"
            )
        if result.stdout:
            self.log.info(result.stdout)
        self._log_run_results()

        # Phase 3 / FRESH-04 / Nyquist dims 4-6: record_run on success path ONLY.
        # NOT reached on failure (AirflowException above) or on skip (AirflowSkipException earlier).
        from aircontroler.hooks.metadata_db import MetadataDBHook
        hook = MetadataDBHook()
        try:
            hook.record_run(
                unique_id=self.unique_id,
                dag_id=context["dag_run"].dag_id,
                run_id=context["run_id"],
            )
        except Exception as exc:
            # Don't fail the task if state recording fails (table missing on Airflow 2.x, etc.)
            # — log WARNING. The model ran successfully; recording is best-effort.
            self.log.warning(
                "record_run failed for %s: %s — state table not updated",
                self.unique_id, exc,
            )
```
